assessment_results
create table public.assessment_results (
  id uuid not null default gen_random_uuid (),
  assessment_id uuid null default gen_random_uuid (),
  father_profile_id uuid null default gen_random_uuid (),
  submitted_at timestamp with time zone null,
  score_total double precision null,
  score_breakdown_json jsonb null,
  interpretation_text character varying null,
  constraint assessment_results_pkey primary key (id),
  constraint assessment_results_assessment_id_fkey foreign KEY (assessment_id) references assessments (id) on update CASCADE on delete CASCADE,
  constraint assessment_results_father_profile_id_fkey foreign KEY (father_profile_id) references father_profiles (id) on update CASCADE on delete CASCADE
) TABLESPACE pg_default;
